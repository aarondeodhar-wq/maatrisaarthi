                  )}
                </div>
              </div>
            </div>
          </div>

          {/* Live Patient Biomarker Simulator */}
          <div className="grid grid-cols-1 lg:grid-cols-12 gap-6">
            
            {/* Input Sliders (7 Cols) */}
            <div className="lg:col-span-7 silky-glass rounded-3xl p-6 shadow-2xl border border-slate-200 dark:border-white/15 space-y-4">
              <h3 className="text-sm font-black text-slate-950 dark:text-white flex items-center justify-between border-b border-slate-200/80 dark:border-white/10 pb-3">
                <span className="flex items-center gap-2">
                  <Activity className="w-4 h-4 text-teal-600 dark:text-teal-400" />
                  <span>Real-Time Biomarker Input Simulator</span>
                </span>
                <span className="text-[10px] font-mono text-slate-600 dark:text-slate-400 font-bold">0ms On-Device Evaluation</span>
              </h3>

              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                {/* Hemoglobin */}
                <div className="bg-slate-50 dark:bg-white/5 border border-slate-200 dark:border-white/10 p-3.5 rounded-2xl space-y-1">
                  <div className="flex justify-between font-bold text-xs">
                    <span className="text-slate-800 dark:text-slate-200">Hemoglobin (Hb):</span>
                    <span className={`font-mono font-black ${hemoglobin < 7 ? 'text-rose-600 dark:text-rose-400 animate-pulse' : 'text-teal-700 dark:text-teal-300'}`}>
                      {hemoglobin} g/dL
                    </span>
                  </div>
                  <input
                    type="range"
                    min="4.0"
                    max="16.0"
                    step="0.1"
                    value={hemoglobin}
                    onChange={e => setHemoglobin(parseFloat(e.target.value))}
                    className="w-full accent-teal-500 cursor-pointer h-2 bg-slate-200 dark:bg-slate-800 rounded-full"
                  />
                </div>

                {/* Blood Pressure */}
                <div className="bg-slate-50 dark:bg-white/5 border border-slate-200 dark:border-white/10 p-3.5 rounded-2xl space-y-1">
                  <div className="flex justify-between font-bold text-xs">
                    <span className="text-slate-800 dark:text-slate-200">Blood Pressure:</span>
                    <span className={`font-mono font-black ${systolicBP >= 140 ? 'text-rose-600 dark:text-rose-400' : 'text-teal-700 dark:text-teal-300'}`}>
                      {systolicBP}/{diastolicBP} mmHg
                    </span>
                  </div>
                  <input
                    type="range"
                    min="80"
                    max="200"
                    step="2"
                    value={systolicBP}
                    onChange={e => setSystolicBP(parseInt(e.target.value))}
                    className="w-full accent-teal-500 cursor-pointer h-2 bg-slate-200 dark:bg-slate-800 rounded-full"
                  />
                </div>

                {/* Blood Glucose */}
                <div className="bg-slate-50 dark:bg-white/5 border border-slate-200 dark:border-white/10 p-3.5 rounded-2xl space-y-1">
                  <div className="flex justify-between font-bold text-xs">
                    <span className="text-slate-800 dark:text-slate-200">Random Blood Sugar:</span>
                    <span className={`font-mono font-black ${bloodSugar >= 140 ? 'text-amber-600 dark:text-amber-300' : 'text-teal-700 dark:text-teal-300'}`}>
                      {bloodSugar} mg/dL
                    </span>
                  </div>
                  <input
                    type="range"
                    min="60"
                    max="250"
                    step="5"
                    value={bloodSugar}
                    onChange={e => setBloodSugar(parseInt(e.target.value))}
                    className="w-full accent-teal-500 cursor-pointer h-2 bg-slate-200 dark:bg-slate-800 rounded-full"
                  />
                </div>

                {/* Maternal Age */}
                <div className="bg-slate-50 dark:bg-white/5 border border-slate-200 dark:border-white/10 p-3.5 rounded-2xl space-y-1">
                  <div className="flex justify-between font-bold text-xs">
                    <span className="text-slate-800 dark:text-slate-200">Maternal Age:</span>
                    <span className={`font-mono font-black ${age < 18 || age > 35 ? 'text-amber-600 dark:text-amber-300' : 'text-teal-700 dark:text-teal-300'}`}>
                      {age} Years
                    </span>
                  </div>
                  <input
                    type="range"
                    min="14"
                    max="45"
                    step="1"
                    value={age}
                    onChange={e => setAge(parseInt(e.target.value))}
                    className="w-full accent-teal-500 cursor-pointer h-2 bg-slate-200 dark:bg-slate-800 rounded-full"
                  />
                </div>
              </div>

              {/* Danger Sign Checkboxes */}
              <div className="bg-slate-50 dark:bg-white/5 border border-slate-200 dark:border-white/10 p-3.5 rounded-2xl space-y-2">
                <span className="text-xs font-black text-slate-900 dark:text-white block">Critical Obstetric Danger Flags:</span>
                <div className="grid grid-cols-2 sm:grid-cols-4 gap-2 text-xs">
                  <label className="flex items-center gap-2 cursor-pointer font-bold text-slate-800 dark:text-slate-200">
                    <input 
                      type="checkbox" 
                      checked={hasBleeding} 
                      onChange={e => setHasBleeding(e.target.checked)}
                      className="accent-rose-500 rounded"
                    />
                    <span className={hasBleeding ? 'text-rose-600 dark:text-rose-400 font-black' : 'text-slate-600 dark:text-slate-400'}>Bleeding</span>
                  </label>

                  <label className="flex items-center gap-2 cursor-pointer font-bold text-slate-800 dark:text-slate-200">
                    <input 
                      type="checkbox" 
                      checked={hasSevereHeadache} 
                      onChange={e => setHasSevereHeadache(e.target.checked)}
                      className="accent-amber-500 rounded"
                    />
                    <span className={hasSevereHeadache ? 'text-amber-600 dark:text-amber-300 font-black' : 'text-slate-600 dark:text-slate-400'}>Headache</span>
                  </label>

                  <label className="flex items-center gap-2 cursor-pointer font-bold text-slate-800 dark:text-slate-200">
                    <input 
                      type="checkbox" 
                      checked={hasFacialEdema} 
                      onChange={e => setHasFacialEdema(e.target.checked)}
                      className="accent-blue-500 rounded"
                    />
                    <span className={hasFacialEdema ? 'text-blue-600 dark:text-blue-300 font-black' : 'text-slate-600 dark:text-slate-400'}>Edema</span>
                  </label>

                  <label className="flex items-center gap-2 cursor-pointer font-bold text-slate-800 dark:text-slate-200">
                    <input 
                      type="checkbox" 
                      checked={hasConvulsions} 
                      onChange={e => setHasConvulsions(e.target.checked)}
                      className="accent-rose-500 rounded"
                    />
                    <span className={hasConvulsions ? 'text-rose-600 dark:text-rose-400 font-black' : 'text-slate-600 dark:text-slate-400'}>Convulsions</span>
                  </label>
                </div>
              </div>
            </div>

            {/* Inference Output Card (5 Cols) */}
            <div className="lg:col-span-5 silky-glass rounded-3xl p-6 shadow-2xl border border-slate-200 dark:border-white/15 space-y-4">
              <h3 className="text-sm font-black text-slate-950 dark:text-white flex items-center justify-between border-b border-slate-200/80 dark:border-white/10 pb-3">
                <span className="flex items-center gap-2">
                  <ShieldCheck className="w-4 h-4 text-teal-600 dark:text-teal-400" />
                  <span>ML Stratification Output</span>
                </span>
                <span className={`text-[10px] font-black px-2.5 py-0.5 rounded-full ${
                  evaluation.riskTier === 'RED' ? 'bg-rose-100 dark:bg-rose-500/20 text-rose-900 dark:text-rose-300 border border-rose-300 dark:border-rose-500/30 animate-pulse' :
                  evaluation.riskTier === 'YELLOW' ? 'bg-amber-100 dark:bg-amber-500/20 text-amber-900 dark:text-amber-300 border border-amber-300 dark:border-amber-500/30' :
                  'bg-emerald-100 dark:bg-emerald-500/20 text-emerald-900 dark:text-emerald-300 border border-emerald-300 dark:border-emerald-500/30'
                }`}>
                  {evaluation.riskTier} RISK
                </span>
              </h3>

              <div className="space-y-3">
                <div className="flex items-center justify-between">
                  <span className="text-xs text-slate-700 dark:text-slate-300 font-bold">Predicted Risk Probability:</span>
                  <span className="text-3xl font-black font-mono text-slate-950 dark:text-white">
                    {(evaluation.riskProbability * 100).toFixed(1)}%
                  </span>
                </div>

                <div className="p-3.5 rounded-2xl bg-slate-100 dark:bg-black/40 border border-slate-200 dark:border-white/10 text-xs text-slate-900 dark:text-slate-200 leading-relaxed font-medium">
                  <strong className="font-black text-slate-950 dark:text-white">Recommended Clinical Action:</strong> {evaluation.triageAction.en}
                </div>

                {/* Top Explainability Drivers */}
                <div className="space-y-1.5 pt-1">
                  <span className="text-[10px] font-black text-slate-700 dark:text-slate-300 uppercase tracking-wider block">
                    Key Feature Attribution:
                  </span>
                  {evaluation.explainability.map((ex, i) => (
                    <div key={i} className="flex items-center justify-between p-2 rounded-xl bg-slate-50 dark:bg-white/5 border border-slate-200 dark:border-white/10 text-xs">
                      <span className="font-bold text-slate-900 dark:text-slate-200">{ex.feature} ({ex.value})</span>
                      <span className={`text-[10px] font-black px-2 py-0.5 rounded-full ${
                        ex.contribution === 'HIGH_RISK' ? 'bg-rose-100 dark:bg-rose-500/20 text-rose-900 dark:text-rose-300 border border-rose-300' :
                        ex.contribution === 'MODERATE_RISK' ? 'bg-amber-100 dark:bg-amber-500/20 text-amber-900 dark:text-amber-300 border border-amber-300' :
                        'bg-emerald-100 dark:bg-emerald-500/20 text-emerald-900 dark:text-emerald-300 border border-emerald-300'
                      }`}>
                        {ex.contribution}
                      </span>
                    </div>
                  ))}
                </div>
              </div>
            </div>
          </div>
        </div>
      )}
    </div>
  );
};
